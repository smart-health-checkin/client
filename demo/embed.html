Embedding <smart-checkin> in any page

This page is deliberately plain: it stands in for your portal. The button below is one custom element plus one script tag — it runs the whole flow (mock wallet + dry-run here) and emits a checkin-complete event. 

<script type="module">
  import { registerScenario } from "";

  // your request, defined by your code — no canned templates
  registerScenario("visit-prep", {
    purpose: "A quick wellness check before today's visit",
    items: [{
      id: "phq2", title: "PHQ-2 questionnaire", required: true,
      content: { kind: "form.fhir",
                 questionnaireCanonical: "https://fhir.loinc.org/Questionnaire/55757-9" },
      accept: ["application/fhir+json"],
    }],
  });
</script>

<smart-checkin
  scenario="visit-prep"
  patient="Patient/example"
  fhir-base="https://hapi.fhir.org/baseR4"
  submit-mode="dry-run"
  mock
  label="Share for check-in">
</smart-checkin>

checkin-complete event payload
